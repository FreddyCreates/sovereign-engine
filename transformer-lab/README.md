## 🚀 Quick Start

### Python
```python
from transformer_lab import UniversalTransformer, TransformerMixer

# Create transformers
phi_t = UniversalTransformer.from_protocol('phi_transformer')
entropy_t = UniversalTransformer.from_protocol('entropy_transformer')

# Mix them
mixer = TransformerMixer()
hybrid = mixer.fuse([phi_t, entropy_t], mode='phi_resonant')

# Process signal
output = hybrid.transform(input_signal)
```

### Rust
```rust
use transformer_lab::{UniversalTransformer, TransformerMixer};

let phi_t = UniversalTransformer::from_protocol("phi_transformer")?;
let entropy_t = UniversalTransformer::from_protocol("entropy_transformer")?;

let mixer = TransformerMixer::new();
let hybrid = mixer.fuse(&[&phi_t, &entropy_t], MixMode::PhiResonant)?;

let output = hybrid.transform(&input_signal)?;
```

### TypeScript
```typescript
import { UniversalTransformer, TransformerMixer } from './transformer-lab';

const phiT = await UniversalTransformer.fromProtocol('phi_transformer');
const entropyT = await UniversalTransformer.fromProtocol('entropy_transformer');

const mixer = new TransformerMixer();
const hybrid = mixer.fuse([phiT, entropyT], { mode: 'phi_resonant' });

const output = hybrid.transform(inputSignal);
```
